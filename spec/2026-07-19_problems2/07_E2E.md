## Task 7.1 — Invoice: Search + Filter + Payment Update E2E (0.3 day)

### `frontend/e2e/invoice.spec.ts` หรือเพิ่มในไฟล์ที่มีอยู่

```ts
test('invoice list — search and filter', async ({ page }) => {
  // Navigate to invoice list
  await page.goto('/sales/invoices');
  await page.waitForSelector('text=ใบแจ้งหนี้');

  // Search by invoice number
  await page.fill('input[type="text"]', 'INV-');
  await page.waitForTimeout(500); // debounce

  // Verify results appear
  await page.waitForSelector('table');

  // Filter by payment status
  await page.click('text=สถานะชำระเงิน');
  await page.click('text=ชำระแล้ว');
  await page.waitForTimeout(300);

  // Verify filter applied
  // ...

  // Filter by payment method
  await page.click('text=วิธีการชำระ');
  await page.click('text=เงินสด');
  await page.waitForTimeout(300);
});

test('invoice detail — update payment status', async ({ page }) => {
  // Open first invoice
  await page.goto('/sales/invoices');
  await page.click('table tr:first-child');
  await page.waitForSelector('text=กลับ');

  // Click update payment button
  await page.click('text=อัพเดทสถานะชำระเงิน');
  await page.waitForSelector('text=สถานะชำระเงิน');

  // Select new status
  await page.click('text=รอชำระ');
  await page.click('text=ชำระแล้ว');
  await page.click('text=บันทึก');

  // Verify success
  await page.waitForTimeout(500);
  await page.waitForSelector('text=ชำระแล้ว');
});
```

## Task 7.3 — Customer Vehicle CRUD E2E (0.3 day)

### `frontend/e2e/customer.spec.ts`

```ts
test('customer detail — add vehicle', async ({ page }) => {
  await page.goto('/customers');
  await page.click('text=สมชาย');
  await page.waitForSelector('text=รถที่ลงทะเบียน');

  // Click add vehicle
  await page.click('text=เพิ่มรถ');
  await page.waitForSelector('text=ทะเบียนรถ');

  // Fill form
  await page.fill('input[placeholder*="ทะเบียน"]', 'กข1234');
  await page.fill('input[placeholder*="ยี่ห้อ"]', 'Toyota');
  await page.fill('input[placeholder*="รุ่น"]', 'Camry');
  await page.click('text=บันทึก');

  // Verify new vehicle appears
  await page.waitForSelector('text=กข1234');
});

test('customer detail — edit vehicle', async ({ page }) => {
  await page.goto('/customers');

  // Click edit on first vehicle
  await page.click('button[title="แก้ไข"]:first-of-type');

  // Update
  await page.fill('input[placeholder*="ยี่ห้อ"]', 'Honda');
```
